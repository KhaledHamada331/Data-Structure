Hash Table (Chaining Implementation)

This implementation uses std::vector of std::list to handle collisions efficiently.

#include <iostream>
#include <vector>
#include <list>

class NeuralHashTable {
    int buckets;
    std::vector<std::list<int>> table;

public:
    NeuralHashTable(int size) : buckets(size) {
        table.resize(buckets);
    }

    // Hash Function: index = key % buckets
    int hashFunction(int key) {
        return key % buckets;
    }

    // Insert key into the hash table
    void insert(int key) {
        int index = hashFunction(key);
        table[index].push_back(key);
    }
};
