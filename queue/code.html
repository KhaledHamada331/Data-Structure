Implementation of a Queue using fixed-size arrays with manual Front and Rear pointer management to demonstrate FIFO logic. 

#include <iostream>
using namespace std;

#define MAX 100

class Queue {
    int front, rear;
public:
    int arr[MAX];

    Queue() { front = 0; rear = 0; } // Initialize pointers

    void enqueue(int x) {
        if (rear == MAX) {
            cout << "Overflow";
            return;
        }
        arr[rear++] = x; // Add at rear and increment
    }

    void dequeue() {
        if (front == rear) {
            cout << "Underflow";
            return;
        }
        // Manual Shifting for demonstration
        for (int i = 0; i < rear - 1; i++) {
            arr[i] = arr[i + 1];
        }
        rear--;
    }

    int getFront() {
        return (front == rear) ? -1 : arr[front];
    }
};
class Queue {
    static final int MAX = 1000;
    int front, rear;
    int a[] = new int[MAX];

    Queue() { front = 0; rear = 0; }

    boolean enqueue(int x) {
        if (rear == MAX) return false;
        a[rear++] = x;
        return true;
    }

    void dequeue() {
        if (front == rear) return;
        for (int i = 0; i < rear - 1; i++) {
            a[i] = a[i + 1];
        }
        rear--;
    }
}
class Queue:
    def __init__(self, size):
        self.size = size
        self.queue = [None] * size
        self.front = 0
        self.rear = 0

    def enqueue(self, item):
        if self.rear == self.size:
            print("Queue Full")
        else:
            self.queue[self.rear] = item
            self.rear += 1

    def dequeue(self):
        if self.front == self.rear:
            print("Queue Empty")
            return None
        # Manual Shifting
        val = self.queue[0]
        for i in range(self.rear - 1):
            self.queue[i] = self.queue[i + 1]
        self.rear -= 1
        return val
